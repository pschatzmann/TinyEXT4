294 bool findPartition(int partition, uint32_t& start) {
295 if (partition < 0 && probe(0)) {
296 start = 0;
297 return true;
298 }
299 uint8_t mbr[66];
300 if (!cache.read(0x1BE, mbr, 66)) return fail(Error::IO);
301 int found = 0;
302 if (mbr[64] == 0x55 && mbr[65] == 0xAA) {
303 for (int j = 0; j < 4; j++) {
304 uint8_t* e = mbr + 16 * j;
305 uint8_t type = e[4];
306 if (type == 0xEE) return findGptPartition(partition, start);
307 uint32_t lba = get32(e + 8);
308 if (type == 0 || lba == 0) continue;
309 if (probe(lba) && (partition < 0 || found++ == partition)) {
310 start = lba;
311 return true;
312 }
313 }
314 }
315 return fail(Error::NotExt4);
316 }

318 bool findGptPartition(int partition, uint32_t& start) {
319 uint8_t hdr[0x58];
320 if (!cache.read(512, hdr, sizeof(hdr))) return fail(Error::IO);
321 if (memcmp(hdr, "EFI PART", 8) != 0) return fail(Error::NotExt4);
322 uint32_t entries = get32(hdr + 0x48);
323 uint32_t count = get32(hdr + 0x50);
324 uint32_t esize = get32(hdr + 0x54);
325 if (count > 128) count = 128;
326 int found = 0;
327 for (uint32_t j = 0; j < count; j++) {
328 uint8_t e[0x28];
329 if (!cache.read((uint64_t)entries * 512 + (uint64_t)j * esize, e, sizeof(e)))
330 return fail(Error::IO);
331 bool used = false;
332 for (int k = 0; k < 16; k++) used |= e[k] != 0;
333 uint32_t lba = get32(e + 0x20);
334 if (!used || lba == 0) continue;
335 if (probe(lba) && (partition < 0 || found++ == partition)) {
336 start = lba;
337 return true;
338 }
339 }
340 return fail(Error::NotExt4);

343 bool readSuper(bool readOnly) {
344 uint8_t sb[0x274];
345 if (!cache.read(ext4::SB_OFFSET, sb, sizeof(sb))) return fail(Error::IO);
346 if (get16(sb + 0x38) != ext4::MAGIC) return fail(Error::NotExt4);
347 incompat = get32(sb + 0x60);
348 ro_compat = get32(sb + 0x64);
349 compat = get32(sb + 0x5C);
350 bool is64 = incompat & ext4::INCOMPAT_64BIT;
351 inodes_count = get32(sb + 0x00);
352 blocks_count = get32(sb + 0x04) | (is64 ? (uint64_t)get32(sb + 0x150) << 32 : 0);
353 free_blocks = get32(sb + 0x0C) | (is64 ? (uint64_t)get32(sb + 0x158) << 32 : 0);
354 free_inodes = get32(sb + 0x10);
355 first_data_block = get32(sb + 0x14);
356 uint32_t log = get32(sb + 0x18);
357 if (log > 6) return fail(Error::Unsupported);
358 log_block = (uint8_t)(10 + log);
359 block_size = 1024u << log;
360 blocks_per_group = get32(sb + 0x20);
361 clusters_per_group = get32(sb + 0x24);
362 inodes_per_group = get32(sb + 0x28);
363 state = get16(sb + 0x3A);
364 uint32_t rev = get32(sb + 0x4C);
365 first_ino = rev ? get32(sb + 0x54) : 11;
366 inode_size = rev ? get16(sb + 0x58) : 128;
367 memcpy(uuid, sb + 0x68, 16);
368 memcpy(volume_name, sb + 0x78, 16);
369 volume_name[16] = 0;
370 reserved_gdt = get16(sb + 0xCE);
371 desc_size = is64 ? get16(sb + 0xFE) : 32;
372 if (desc_size < 32) desc_size = 32;
373 first_meta_bg = get32(sb + 0x104);
374 backup_bgs[0] = get32(sb + 0x24C);
375 backup_bgs[1] = get32(sb + 0x250);
376 if (incompat & ext4::INCOMPAT_CSUM_SEED)
377 csum_seed = get32(sb + 0x270);
378 else
379 csum_seed = crc32c(0xFFFFFFFF, uuid, 16);
380
381 if (blocks_per_group == 0 || inodes_per_group == 0 ||
382 blocks_per_group > block_size * 8 || inodes_per_group > block_size * 8 ||
383 inode_size < 128 || inode_size > block_size || desc_size > 64 ||
384 first_data_block >= blocks_count)
385 return fail(Error::Corrupt);
386 if (incompat & ~ext4::INCOMPAT_READ) return fail(Error::Unsupported);
387 group_count = (uint32_t)((blocks_count - first_data_block + blocks_per_group - 1) /
388 blocks_per_group);
389
390 bool canWrite = TINYEXT4_WRITE &&
391 (incompat & ~ext4::INCOMPAT_WRITE) == 0 &&
392 (ro_compat & ~ext4::RO_COMPAT_WRITE) == 0 &&
393 (incompat & ext4::INCOMPAT_EXTENTS) &&
394 (incompat & ext4::INCOMPAT_FILETYPE) &&
395 (state & ext4::STATE_VALID) && !(state & ext4::STATE_ERROR);
396 read_only = readOnly || !canWrite;
397 if (!readOnly && read_only) {
398 TINYEXT4_LOGF("tinyext4: mounted read only (features 0x%x/0x%x, state %d)\n",
399 (unsigned)incompat, (unsigned)ro_compat, state);
400 }
401 return true;
402 }

404 bool hasSuper(uint32_t g) const {
405 if (g == 0) return true;
406 if (compat & ext4::COMPAT_SPARSE_SUPER2)
407 return g == backup_bgs[0] || g == backup_bgs[1];
408 if (g == 1 || !(ro_compat & ext4::RO_COMPAT_SPARSE_SUPER)) return true;
409 if (!(g & 1)) return false;
410 static const uint8_t bases[3] = {3, 5, 7};
411 for (uint32_t b : bases) {
412 uint32_t n = g;
413 while (n % b == 0) n /= b;
414 if (n == 1) return true;
415 }
416 return false;
417 }

423 uint64_t descOffset(uint32_t g) const {
424 uint32_t per = block_size / desc_size;
425 uint32_t db = g / per;
426 uint64_t blk;
427 if (!(incompat & ext4::INCOMPAT_META_BG) || db < first_meta_bg) {
428 blk = first_data_block + 1 + db;
429 } else {
430 uint32_t first = db * per;
431 blk = groupFirstBlock(first) + (hasSuper(first) ? 1 : 0);
432 }
433 return blockOff(blk) + (uint64_t)(g % per) * desc_size;
434 }

442 bool readDesc(uint32_t g, GroupDesc& d) {
443 uint8_t raw[64] = {0};
444 if (g >= group_count) return fail(Error::Corrupt);
445 if (!cache.read(descOffset(g), raw, desc_size)) return fail(Error::IO);
446 bool big = desc_size >= 64;
447 d.blockBitmap = get32(raw + 0x00) | (big ? (uint64_t)get32(raw + 0x20) << 32 : 0);
448 d.inodeBitmap = get32(raw + 0x04) | (big ? (uint64_t)get32(raw + 0x24) << 32 : 0);
449 d.inodeTable = get32(raw + 0x08) | (big ? (uint64_t)get32(raw + 0x28) << 32 : 0);
450 d.freeBlocks = get16(raw + 0x0C) | (big ? (uint32_t)get16(raw + 0x2C) << 16 : 0);
451 d.freeInodes = get16(raw + 0x0E) | (big ? (uint32_t)get16(raw + 0x2E) << 16 : 0);
452 d.usedDirs = get16(raw + 0x10) | (big ? (uint32_t)get16(raw + 0x30) << 16 : 0);
453 d.flags = hasGroupCsum() ? get16(raw + 0x12) : 0;
454 d.itableUnused = get16(raw + 0x1C) | (big ? (uint32_t)get16(raw + 0x32) << 16 : 0);
455 return true;
456 }

503 bool markGroup(uint32_t g, uint8_t mask) {
504 sb_dirty = true;
505 if (!hasCsum()) return true;
506 for (int j = 0; j < dirty_group_count; j++) {
507 if (dirty_groups[j].group == g) {
508 dirty_groups[j].mask |= mask;
509 return true;
510 }
511 }
512 if (dirty_group_count == TINYEXT4_DIRTY_GROUPS && !syncGroups()) return false;
513 dirty_groups[dirty_group_count++] = {g, mask};
514 return true;
515 }

518 bool syncGroups() {
519 for (int j = 0; j < dirty_group_count; j++) {
520 uint32_t g = dirty_groups[j].group;
521 GroupDesc d;
522 uint8_t raw[64] = {0};
523 if (!readDesc(g, d)) return false;
524 uint64_t off = descOffset(g);
525 if (!cache.read(off, raw, desc_size)) return fail(Error::IO);
526 if ((dirty_groups[j].mask & DIRTY_BLOCKS) && !(d.flags & ext4::BG_BLOCK_UNINIT)) {
527 uint32_t crc = csum_seed;
528 if (!cache.crc32c(blockOff(d.blockBitmap), clusters_per_group / 8, crc))
529 return fail(Error::IO);
530 put16(raw + 0x18, (uint16_t)crc);
531 if (desc_size >= 0x3C) put16(raw + 0x38, (uint16_t)(crc >> 16));
532 }
533 if ((dirty_groups[j].mask & DIRTY_INODES) && !(d.flags & ext4::BG_INODE_UNINIT)) {
534 uint32_t crc = csum_seed;
535 if (!cache.crc32c(blockOff(d.inodeBitmap), inodes_per_group / 8, crc))
536 return fail(Error::IO);
537 put16(raw + 0x1A, (uint16_t)crc);
538 if (desc_size >= 0x3C) put16(raw + 0x3A, (uint16_t)(crc >> 16));
539 }
540 descChecksum(g, raw);
541 if (!cache.write(off, raw, desc_size)) return fail(Error::IO);
542 }
543 dirty_group_count = 0;
544 return true;
545 }

547 bool writeSuper() {
548 uint64_t sb = ext4::SB_OFFSET;
549 bool ok = cache.write32(sb + 0x0C, (uint32_t)free_blocks) &&
550 cache.write32(sb + 0x10, free_inodes) &&
551 cache.write32(sb + 0x30, now()) && cache.write16(sb + 0x3A, state);
552 if (ok && (incompat & ext4::INCOMPAT_64BIT))
553 ok = cache.write32(sb + 0x158, (uint32_t)(free_blocks >> 32));
554 if (ok && hasCsum()) {
555 uint32_t crc = 0xFFFFFFFF;
556 ok = cache.crc32c(sb, ext4::SB_CSUM_OFFSET, crc) &&
557 cache.write32(sb + ext4::SB_CSUM_OFFSET, crc);
558 }
559 if (!ok) return fail(Error::IO);
560 sb_dirty = false;
561 return true;
562 }

581 bool changeBits(uint64_t bmOff, uint32_t bit, uint32_t count, bool value,
582 uint32_t& changed) {
583 changed = 0;
584 while (count > 0) {
585 uint64_t off = bmOff + bit / 8;
586 uint8_t b;
587 if ((bit & 7) == 0 && count >= 8) {
588 // whole bytes
589 uint8_t buf[32];
590 uint32_t n = count / 8 > sizeof(buf) ? sizeof(buf) : count / 8;
591 if (!cache.read(off, buf, n)) return fail(Error::IO);
592 for (uint32_t j = 0; j < n; j++) {
593 uint8_t nb = value ? 0xFF : 0;
594 for (int k = 0; k < 8; k++) changed += ((buf[j] >> k) & 1) != value;
595 buf[j] = nb;
596 }
597 if (!cache.write(off, buf, n)) return fail(Error::IO);
598 bit += n * 8;
599 count -= n * 8;
600 continue;
601 }
602 if (!cache.read(off, &b, 1)) return fail(Error::IO);
603 uint8_t mask = (uint8_t)(1 << (bit & 7));
604 if (((b & mask) != 0) != value) {
605 changed++;
606 b = value ? (b | mask) : (b & ~mask);
607 if (!cache.write(off, &b, 1)) return fail(Error::IO);
608 }
609 bit++;
610 count--;
611 }
612 return true;
613 }

621 bool findZero(uint64_t bmOff, uint32_t from, uint32_t to, uint32_t& bit) {
622 uint8_t buf[32];
623 uint32_t j = from;
624 while (j < to) {
625 uint32_t byte = j / 8;
626 uint32_t n = (to + 7) / 8 - byte;
627 if (n > sizeof(buf)) n = sizeof(buf);
628 if (!cache.read(bmOff + byte, buf, n)) return fail(Error::IO);
629 for (uint32_t k = 0; k < n && j < to; k++) {
630 if (buf[k] == 0xFF) {
631 j = (byte + k + 1) * 8;
632 continue;
633 }
634 for (; j < (byte + k + 1) * 8 && j < to; j++) {
635 if (!(buf[k] & (1 << (j & 7)))) {
636 bit = j;
637 return true;
638 }
639 }
640 }
641 }
642 return false;

646 bool initBlockBitmap(uint32_t g, const GroupDesc& d) {
647 if (incompat & ext4::INCOMPAT_META_BG) return fail(Error::Unsupported);
648 uint64_t bm = blockOff(d.blockBitmap);
649 uint32_t n = blocksInGroup(g);
650 if (!cache.fill(bm, 0, block_size)) return fail(Error::IO);
651 if (hasSuper(g)) {
652 uint32_t per = block_size / desc_size;
653 uint32_t gdt = (group_count + per - 1) / per;
654 if (!setBits(bm, 0, 1 + gdt + reserved_gdt)) return false;
655 }
656 // bitmaps and inode tables (flex_bg: of any group) located in this group
657 uint64_t first = groupFirstBlock(g);
658 uint64_t last = first + n;
659 uint32_t itBlocks =
660 (uint32_t)(((uint64_t)inodes_per_group * inode_size + block_size - 1) / block_size);
661 for (uint32_t h = 0; h < group_count; h++) {
662 GroupDesc o;
663 if (!readDesc(h, o)) return false;
664 uint64_t starts[3] = {o.blockBitmap, o.inodeBitmap, o.inodeTable};
665 uint32_t lens[3] = {1, 1, itBlocks};
666 for (int k = 0; k < 3; k++) {
667 uint64_t s = starts[k], e = starts[k] + lens[k];
668 if (e <= first || s >= last) continue;
669 if (s < first) s = first;
670 if (e > last) e = last;
671 if (!setBits(bm, (uint32_t)(s - first), (uint32_t)(e - s))) return false;
672 }
673 }
674 return setBits(bm, n, block_size * 8 - n);
675 }

678 uint64_t bm = blockOff(d.inodeBitmap);
679 if (!cache.fill(bm, 0, block_size)) return fail(Error::IO);
680 return setBits(bm, inodes_per_group, block_size * 8 - inodes_per_group);
681 }

684 bool allocBlock(uint64_t goal, uint64_t& out) {
685 if (free_blocks == 0) return fail(Error::NoSpace);
686 uint32_t g0 = 0, startBit = 0;
687 if (goal >= first_data_block && goal < blocks_count) {
688 g0 = (uint32_t)((goal - first_data_block) / blocks_per_group);
689 startBit = (uint32_t)((goal - first_data_block) % blocks_per_group);
690 }
691 for (uint32_t i = 0; i <= group_count; i++) {
692 uint32_t g = (g0 + i) % group_count;
693 GroupDesc d;
694 if (!readDesc(g, d)) return false;
695 if (d.freeBlocks == 0) continue;
696 if (d.flags & ext4::BG_BLOCK_UNINIT) {
697 if (!initBlockBitmap(g, d)) {
698 if (error == Error::Unsupported) continue;
699 return false;
700 }
701 d.flags &= ~ext4::BG_BLOCK_UNINIT;
702 }
703 uint32_t bit;
704 uint64_t bm = blockOff(d.blockBitmap);
705 if (!findZero(bm, i == 0 ? startBit : 0, blocksInGroup(g), bit)) {
706 if (error == Error::IO) return false;
707 continue;
708 }
709 if (!setBits(bm, bit, 1)) return false;
710 d.freeBlocks--;
711 if (!writeDesc(g, d) || !markGroup(g, DIRTY_BLOCKS)) return false;
712 free_blocks--;
713 out = groupFirstBlock(g) + bit;
714 return true;
715 }
716 return fail(Error::NoSpace);
717 }

720 bool freeBlockRange(uint64_t start, uint64_t count) {
721 while (count > 0) {
722 if (start < first_data_block || start + count > blocks_count)
723 return fail(Error::Corrupt);
724 uint32_t g = (uint32_t)((start - first_data_block) / blocks_per_group);
725 uint32_t bit = (uint32_t)((start - first_data_block) % blocks_per_group);
726 uint32_t n = blocks_per_group - bit;
727 if (n > count) n = (uint32_t)count;
728 GroupDesc d;
729 if (!readDesc(g, d)) return false;
730 if (d.flags & ext4::BG_BLOCK_UNINIT) return fail(Error::Corrupt);
731 uint32_t changed;
732 if (!changeBits(blockOff(d.blockBitmap), bit, n, false, changed)) return false;
733 d.freeBlocks += changed;
734 if (!writeDesc(g, d) || !markGroup(g, DIRTY_BLOCKS)) return false;
735 free_blocks += changed;
736 start += n;
737 count -= n;
738 }
739 return true;
740 }

756 bool allocInode(uint32_t hint, bool isDir, uint32_t& ino) {
757 if (free_inodes == 0) return fail(Error::NoSpace);
758 for (uint32_t i = 0; i < group_count; i++) {
759 uint32_t g = (hint + i) % group_count;
760 GroupDesc d;
761 if (!readDesc(g, d)) return false;
762 if (d.freeInodes == 0) continue;
763 if (d.flags & ext4::BG_BLOCK_UNINIT) {
764 // like the kernel: initialize the block bitmap of the group as well
765 if (!initBlockBitmap(g, d)) {
766 if (error == Error::Unsupported) continue;
767 return false;
768 }
769 d.flags &= ~ext4::BG_BLOCK_UNINIT;
770 if (!markGroup(g, DIRTY_BLOCKS)) return false;
771 }
772 if (d.flags & ext4::BG_INODE_UNINIT) {
773 if (!initInodeBitmap(d)) return false;
774 d.flags &= ~ext4::BG_INODE_UNINIT;
775 }
776 uint32_t bit;
777 uint64_t bm = blockOff(d.inodeBitmap);
778 uint32_t from = g == 0 ? first_ino - 1 : 0;
779 if (!findZero(bm, from, inodes_per_group, bit)) {
780 if (error == Error::IO) return false;
781 continue;
782 }
783 if (!setBits(bm, bit, 1)) return false;
784 d.freeInodes--;
785 if (isDir) d.usedDirs++;
786 if (hasGroupCsum() && bit >= inodes_per_group - d.itableUnused)
787 d.itableUnused = inodes_per_group - bit - 1;
788 if (!writeDesc(g, d) || !markGroup(g, DIRTY_INODES)) return false;
789 free_inodes--;
790 ino = g * inodes_per_group + bit + 1;
791 return true;
792 }
793 return fail(Error::NoSpace);
794 }

842 if (in.flags & ext4::FL_HUGE_FILE) in.blocks <<= (log_block - 9);
843 memcpy(in.block, raw + 0x28, 60);
844 in.generation = get32(raw + 0x64);

881 bool inodeChecksum(uint32_t ino, uint32_t gen, uint64_t off) {
882 if (!hasCsum()) return true;
883 bool hasHi = false;
884 if (inode_size > 128) {
885 uint8_t e[2];
886 if (!cache.read(off + 0x80, e, 2)) return fail(Error::IO);
887 hasHi = get16(e) >= 4;

890 uint8_t buf[32];
891 for (uint32_t pos = 0; pos < inode_size; pos += sizeof(buf)) {
892 if (!cache.read(off + pos, buf, sizeof(buf))) return fail(Error::IO);
893 for (uint32_t k = 0; k < sizeof(buf); k++) {
894 uint32_t a = pos + k;
895 if (a == 0x7C || a == 0x7D || (hasHi && (a == 0x82 || a == 0x83))) buf[k] = 0;
896 }
897 crc = crc32c(crc, buf, sizeof(buf));
898 }
899 if (!hasHi) crc &= 0xFFFF;
900 if (!cache.write16(off + 0x7C, (uint16_t)crc)) return fail(Error::IO);
901 if (hasHi && !cache.write16(off + 0x82, (uint16_t)(crc >> 16))) return fail(Error::IO);
902 return true;
903 }

961 bool mapBlock(Inode& in, uint32_t lblk, uint64_t& pblk, uint32_t& run,
962 bool* uninit = nullptr) {
963 if (uninit) *uninit = false;
964 pblk = 0;
965 run = 1;
966 if (!in.hasExtents()) return mapIndirect(in, lblk, pblk);
967 Node n{0, in.block};
968 uint8_t h[12], e[12];
969 if (!nodeGet(n, 0, h)) return false;
970 for (int level = 0; level < 8; level++) {
971 if (get16(h) != ext4::EXT_MAGIC) return fail(Error::Corrupt);
972 uint16_t entries = get16(h + 2);
973 if (get16(h + 6) == 0) {
974 for (uint32_t i = 1; i <= entries; i++) {
975 if (!nodeGet(n, i, e)) return false;
976 uint32_t start = get32(e);
977 uint32_t len = get16(e + 4);
978 bool un = len > ext4::EXT_INIT_MAX_LEN;
979 if (un) len -= ext4::EXT_INIT_MAX_LEN;
980 if (lblk < start) {
981 run = start - lblk;
982 return true;
983 }
984 if (lblk < start + len) {
985 run = start + len - lblk;
986 if (un) {
987 if (uninit) *uninit = true;
988 } else {
989 pblk = extStart(e) + (lblk - start);
990 }
991 return true;
992 }
993 }
994 run = 0xFFFFFFFF - lblk;
995 return true;
996 }
997 uint64_t child = 0;
998 uint32_t next = 0xFFFFFFFF;
999 for (uint32_t i = 1; i <= entries; i++) {
1000 if (!nodeGet(n, i, e)) return false;
1001 if (get32(e) > lblk) {
1002 next = get32(e);
1003 break;
1004 }
1005 child = idxLeaf(e);
1006 }
1007 if (child == 0) {
1008 run = next - lblk;
1009 return true;
1010 }
1011 n = Node{child, nullptr};
1012 if (!nodeGet(n, 0, h)) return false;
1013 }
1014 return fail(Error::Corrupt);
1015 }

1017 bool mapIndirect(Inode& in, uint32_t lblk, uint64_t& pblk) {
1018 uint32_t apb = block_size / 4;
1019 if (lblk < 12) {
1020 pblk = get32(in.block + 4 * lblk);
1021 return true;
1022 }
1023 lblk -= 12;
1024 int level, slot;
1025 if (lblk < apb) {
1026 level = 1;
1027 slot = 12;
1028 } else if ((lblk -= apb) < (uint64_t)apb * apb) {
1029 level = 2;
1030 slot = 13;
1031 } else {
1032 lblk -= apb * apb;
1033 level = 3;
1034 slot = 14;
1035 }
1036 uint32_t ptr = get32(in.block + 4 * slot);
1037 while (level > 0 && ptr != 0) {
1038 uint32_t div = 1;
1039 for (int j = 1; j < level; j++) div *= apb;
1040 if (!cache.read32(blockOff(ptr) + 4 * (lblk / div), ptr)) return fail(Error::IO);
1041 lblk %= div;
1042 level--;
1043 }
1044 pblk = ptr;
1045 return true;
1046 }

1161 if (!nodeGet(path[0], 0, hdr[0])) return false;

1164 uint8_t e[12];

1166 if (entries == 0) return fail(Error::Corrupt);
1167 if (!nodeGet(path[lvl], entries, e)) return false;

1170 if (!nodeGet(path[lvl], 0, hdr[lvl])) return false;
1171 }
1172 if (get16(hdr[lvl]) != ext4::EXT_MAGIC) return fail(Error::Corrupt);
1173 Node& leaf = path[lvl];
1174 uint16_t entries = get16(hdr[lvl] + 2);
1175 uint16_t max = get16(hdr[lvl] + 4);
1176 uint8_t e[12];

1181 if (len > ext4::EXT_INIT_MAX_LEN) {
1182 if (lblk < start + len - ext4::EXT_INIT_MAX_LEN) return fail(Error::Unsupported);
1183 } else {
1184 if (lblk < start + len) return fail(Error::Unsupported);
1185 if (start + len == lblk && extStart(e) + len == pblk &&
1186 len < ext4::EXT_INIT_MAX_LEN) {
1187 put16(e + 4, len + 1);
1188 return nodePut(in, leaf, entries, e);
1189 }
1190 }
1191 }
1192 makeExtent(e, lblk, 1, pblk);
1193 if (entries < max) {
1194 put16(hdr[lvl] + 2, entries + 1);
1195 return nodePut(in, leaf, entries + 1, e) && nodePut(in, leaf, 0, hdr[lvl]);
1196 }
1197 // leaf is full: find the deepest index node with a free slot
1198 int k = lvl - 1;
1199 while (k >= 0 && get16(hdr[k] + 2) >= get16(hdr[k] + 4)) k--;
1200 if (k < 0) {
1201 if (!growTree(in, pblk)) return false;
1202 continue; // retry with the new root
1203 }
1204 // new chain of nodes below level k, bottom up
1205 uint64_t child;
1206 if (!newTreeBlock(in, pblk + 1, 0, child)) return false;
1207 Node n{child, nullptr};
1208 uint8_t h[12];
1209 makeHeader(h, 1, blockMaxEntries(), 0);
1210 if (!nodePut(in, n, 1, e) || !nodePut(in, n, 0, h)) return false;
1211 for (int level = lvl - 1; level > k; level--) {
1212 uint64_t parent;
1213 if (!newTreeBlock(in, pblk + 1, (uint16_t)(lvl - level), parent)) return false;
1214 Node p{parent, nullptr};
1215 uint8_t ie[12];
1216 makeIndex(ie, lblk, child);
1217 makeHeader(h, 1, blockMaxEntries(), (uint16_t)(lvl - level));
1218 if (!nodePut(in, p, 1, ie) || !nodePut(in, p, 0, h)) return false;
1219 child = parent;
1220 }
1221 uint8_t ie[12];
1222 makeIndex(ie, lblk, child);
1223 uint16_t ke = get16(hdr[k] + 2) + 1;
1224 put16(hdr[k] + 2, ke);
1225 return nodePut(in, path[k], ke, ie) && nodePut(in, path[k], 0, hdr[k]);
1226 }
1227 return fail(Error::Corrupt);
1228 }

1231 bool growTree(Inode& in, uint64_t goal) {
1232 uint8_t h[12];
1233 memcpy(h, in.block, 12);
1234 uint16_t depth = get16(h + 6);
1235 uint16_t entries = get16(h + 2);
1236 if (depth >= 4) return fail(Error::Unsupported);
1237 uint64_t blk;
1238 if (!newTreeBlock(in, goal, depth, blk)) return false;
1239 uint64_t off = blockOff(blk);
1240 makeHeader(h, entries, blockMaxEntries(), depth);
1241 if (!cache.write(off, h, 12) || !cache.write(off + 12, in.block + 12, 12 * entries))
1242 return fail(Error::IO);
1243 uint8_t ie[12];
1244 makeIndex(ie, entries > 0 ? get32(in.block + 12) : 0, blk);
1245 makeHeader(h, 1, 4, depth + 1);
1246 memset(in.block, 0, 60);
1247 memcpy(in.block, h, 12);
1248 memcpy(in.block + 12, ie, 12);
1249 return true;
1250 }

1253 bool truncNode(Inode& in, const Node& n, uint32_t cutoff, bool& empty) {
1254 uint8_t h[12], e[12];
1255 if (!nodeGet(n, 0, h)) return false;
1256 if (get16(h) != ext4::EXT_MAGIC) return fail(Error::Corrupt);
1257 uint16_t entries = get16(h + 2);
1258 uint16_t old = entries;
1259 if (get16(h + 6) == 0) {
1260 while (entries > 0) {
1261 if (!nodeGet(n, entries, e)) return false;
1262 uint32_t start = get32(e);
1263 uint16_t raw = get16(e + 4);
1264 bool un = raw > ext4::EXT_INIT_MAX_LEN;
1265 uint32_t len = un ? raw - ext4::EXT_INIT_MAX_LEN : raw;
1266 if (start >= cutoff) {
1267 if (!freeInodeBlocks(in, extStart(e), len)) return false;
1268 entries--;
1269 continue;
1270 }
1271 if (start + len > cutoff) {
1272 uint32_t keep = cutoff - start;
1273 if (!freeInodeBlocks(in, extStart(e) + keep, len - keep)) return false;
1274 put16(e + 4, (uint16_t)(un ? keep + ext4::EXT_INIT_MAX_LEN : keep));
1275 if (!nodePut(in, n, entries, e)) return false;
1276 }
1277 break;
1278 }
1279 } else {
1280 while (entries > 0) {
1281 if (!nodeGet(n, entries, e)) return false;
1282 uint32_t start = get32(e);
1283 uint64_t child = idxLeaf(e);
1284 bool childEmpty = false;
1285 if (!truncNode(in, Node{child, nullptr}, cutoff, childEmpty)) return false;
1286 if (childEmpty) {
1287 if (!freeInodeBlocks(in, child, 1)) return false;
1288 entries--;
1289 if (start >= cutoff) continue;
1290 }
1291 break;
1292 }
1293 }
1294 if (entries != old) {
1295 put16(h + 2, entries);
1296 if (!nodePut(in, n, 0, h)) return false;
1297 }
1298 empty = entries == 0;
1299 return true;
1300 }

1316 bool truncateInode(Inode& in, uint64_t size) {
1317 if (in.isInline()) {
1318 if (size > in.size || size > 60) return fail(Error::Unsupported);
1319 memset(in.block + size, 0, 60 - size);
1320 in.size = size;
1321 return true;
1322 }
1323 if (hasDataBlocks(in)) {
1324 uint32_t cutoff = (uint32_t)((size + block_size - 1) >> log_block);
1325 if (in.hasExtents()) {
1326 bool empty;
1327 if (!truncNode(in, Node{0, in.block}, cutoff, empty)) return false;
1328 if (empty) makeHeader(in.block, 0, 4, 0);
1329 } else {
1330 if (size != 0) return fail(Error::Unsupported);
1331 for (int j = 0; j < 12; j++)
1332 if (!freeIndirect(in, get32(in.block + 4 * j), 0)) return false;
1333 for (int j = 12; j < 15; j++)
1334 if (!freeIndirect(in, get32(in.block + 4 * j), j - 11)) return false;
1335 memset(in.block, 0, 60);
1336 }
1337 // zero the rest of the last block, so that growing the file reads zeros
1338 uint32_t rest = (uint32_t)(size & (block_size - 1));
1339 if (rest > 0 && size < in.size) {
1340 uint64_t pblk;
1341 uint32_t run;
1342 if (!mapBlock(in, (uint32_t)(size >> log_block), pblk, run)) return false;
1343 if (pblk && !cache.fill(blockOff(pblk) + rest, 0, block_size - rest))
1344 return fail(Error::IO);
1345 }
1346 }
1347 in.size = size;
1348 return true;
1349 }

1353 if (in.fileAcl == 0) return true;
1354 uint64_t off = blockOff(in.fileAcl);
1355 uint8_t h[8];

1358 if (get32(h) == 0xEA020000 && refs > 1) {
1359 if (!cache.write32(off + 4, refs - 1)) return fail(Error::IO);
1360 if (hasCsum()) {

1362 put32(nr, (uint32_t)in.fileAcl);

1364 uint32_t crc = crc32c(csum_seed, nr, 8);
1365 uint8_t zero[4] = {0, 0, 0, 0};

1367 crc = crc32c(crc, zero, 4);
1368 if (!cache.crc32c(off + 0x14, block_size - 0x14, crc) ||
1369 !cache.write32(off + 0x10, crc))
1370 return fail(Error::IO);
1371 }
1372 } else if (!freeInodeBlocks(in, in.fileAcl, 1)) {
1373 return false;
1374 }
1375 in.fileAcl = 0;
1376 return true;
1377 }

1393 static uint32_t dirRecLen(uint32_t nameLen) { return (8 + nameLen + 3) & ~3u; }
1394 uint32_t decodeRecLen(uint16_t v) const {
1395 if (block_size < 65536) return v;
1396 if (v == 65535 || v == 0) return block_size;
1397 return (v & 65532) | ((v & 3) << 16);
1398 }
1399 uint16_t encodeRecLen(uint32_t len) const {
1400 if (block_size < 65536) return (uint16_t)len;
1401 if (len == 65536) return 65535;
1402 return (uint16_t)((len & 65532) | ((len >> 16) & 3));
1403 }

1406 bool dirNextRaw(Inode& dir, DirIter& it, RawDirent& d) {
1407 while (it.pos < dir.size) {
1408 uint32_t lblk = (uint32_t)(it.pos >> log_block);
1409 uint32_t off = (uint32_t)(it.pos & (block_size - 1));
1410 uint64_t pblk;
1411 uint32_t run;
1412 if (!cachedMap(dir, it.map, lblk, pblk, run)) return false;
1413 if (pblk == 0) {
1414 it.pos = (uint64_t)(lblk + 1) << log_block;
1415 continue;
1416 }
1417 uint8_t h[8];
1418 d.dev = blockOff(pblk) + off;
1419 if (!cache.read(d.dev, h, 8)) return fail(Error::IO);
1420 d.inode = get32(h);
1421 d.recLen = decodeRecLen(get16(h + 4));
1422 d.nameLen = h[6];
1423 d.type = hasFiletype() ? h[7] : 0;
1424 if (d.recLen < 8 || (d.recLen & 3) || off + d.recLen > block_size ||
1425 d.nameLen + 8u > d.recLen) {
1426 // damaged block: skip it
1427 it.pos = (uint64_t)(lblk + 1) << log_block;
1428 continue;
1429 }
1430 it.pos += d.recLen;
1431 return true;
1432 }
1433 return false;
1434 }

1442 if (memcmp(buf, name + pos, n) != 0) return false;

1465 bool resolve(const char* path, size_t len, Inode& out) {
1466 if (!mounted) return fail(Error::NotMounted);
1467 if (!readInode(ext4::ROOT_INO, out)) return false;
1468 size_t pos = 0;
1469 while (pos < len) {
1470 while (pos < len && path[pos] == '/') pos++;
1471 size_t start = pos;
1472 while (pos < len && path[pos] != '/') pos++;
1473 size_t n = pos - start;
1474 if (n == 0) break;
1475 if (n > 255) return fail(Error::Invalid);
1476 uint32_t ino;
1477 if (!lookup(out, path + start, n, ino)) return false;
1478 if (!readInode(ino, out)) return false;
1479 }
1480 return true;

1484 bool resolveParent(const char* path, Inode& parent, const char*& name,
1485 uint8_t& nameLen) {
1486 size_t len = strlen(path);
1487 while (len > 0 && path[len - 1] == '/') len--;
1488 size_t start = len;
1489 while (start > 0 && path[start - 1] != '/') start--;
1490 size_t n = len - start;
1491 if (n == 0 || n > 255 || (n == 1 && path[start] == '.') ||
1492 (n == 2 && path[start] == '.' && path[start + 1] == '.'))
1493 return fail(Error::Invalid);
1494 name = path + start;
1495 nameLen = (uint8_t)n;
1496 if (!resolve(path, start, parent)) return false;
1497 if (!parent.isDir()) return fail(Error::NotDir);
1498 return true;

1598 bool dirAdd(Inode& dir, const char* name, uint8_t len, uint32_t ino, FileType type) {
1599 if (dir.isInline()) return fail(Error::Unsupported);
1600 if ((dir.flags & ext4::FL_INDEX) && !dropHtree(dir)) return false;
1601 uint32_t need = dirRecLen(len);
1602 uint32_t usable = block_size - (hasCsum() ? ext4::DIR_TAIL_SIZE : 0);
1603 uint32_t blocks = (uint32_t)(dir.size >> log_block);
1604 uint64_t last = 0;
1605 for (uint32_t b = 0; b < blocks; b++) {
1606 uint64_t pblk;
1607 uint32_t run;
1608 if (!mapBlock(dir, b, pblk, run)) return false;
1609 if (pblk == 0) continue;
1610 last = pblk;
1611 uint64_t base = blockOff(pblk);
1612 uint32_t off = 0;
1613 while (off < usable) {
1614 uint8_t h[8];
1615 if (!cache.read(base + off, h, 8)) return fail(Error::IO);
1616 uint32_t entIno = get32(h);
1617 uint32_t rec = decodeRecLen(get16(h + 4));
1618 if (rec < 8 || off + rec > block_size) return fail(Error::Corrupt);
1619 uint32_t used = entIno ? dirRecLen(h[6]) : 0;
1620 uint32_t limit = off + rec > usable ? usable - off : rec;
1621 if (limit >= used + need) {
1622 if (entIno == 0) {
1623 if (!writeDirent(base + off, ino, rec, name, len, type)) return false;
1624 } else {
1625 if (!cache.write16(base + off + 4, encodeRecLen(used)) ||
1626 !writeDirent(base + off + used, ino, rec - used, name, len, type))
1627 return fail(Error::IO);
1628 }
1629 return dirBlockCsum(dir, pblk);
1630 }
1631 off += rec;
1632 }
1633 }
1634 // no space: append a new block
1635 uint64_t pblk;
1636 if (!allocBlock(last ? last + 1 : groupFirstBlock((dir.ino - 1) / inodes_per_group), pblk))
1637 return false;
1638 if (!appendExtent(dir, blocks, pblk)) {
1639 freeBlockRange(pblk, 1);
1640 return false;
1641 }
1642 dir.blocks += sectorsPerBlock();
1643 dir.size += block_size;
1644 uint64_t base = blockOff(pblk);
1645 if (!cache.fill(base, 0, block_size)) return fail(Error::IO);
1646 if (!writeDirent(base, ino, usable, name, len, type)) return false;
1647 if (hasCsum() && !writeDirTail(base)) return false;
1648 return dirBlockCsum(dir, pblk) && writeInode(dir);
1649 }

1752 size_t fileRead(File& f, uint8_t* buf, size_t len) {
1753 Inode& in = f.inode;
1754 size_t done = 0;
1755 if (in.isDir()) {
1756 fail(Error::IsDir);
1757 return 0;
1758 }
1759 if (in.isInline() || (in.isSymlink() && !hasDataBlocks(in))) {
1760 // data in i_block (inline data beyond 60 bytes in xattrs is not supported)
1761 while (done < len && f.pos < in.size && f.pos < 60) buf[done++] = in.block[f.pos++];
1762 return done;
1763 }
1764 while (done < len && f.pos < in.size) {
1765 uint32_t lblk = (uint32_t)(f.pos >> log_block);
1766 uint32_t off = (uint32_t)(f.pos & (block_size - 1));
1767 uint64_t pblk;
1768 uint32_t run;
1769 if (!cachedMap(in, f.map, lblk, pblk, run)) break;
1770 uint64_t n = ((uint64_t)run << log_block) - off;
1771 if (n > len - done) n = len - done;
1772 if (n > in.size - f.pos) n = in.size - f.pos;
1773 if (pblk == 0) {
1774 memset(buf + done, 0, (size_t)n);
1775 } else if (!cache.readBulk(blockOff(pblk) + off, buf + done, (size_t)n)) {
1776 fail(Error::IO);
1777 break;
1778 }
1779 done += (size_t)n;
1780 f.pos += n;
1781 }
1782 return done;
1783 }

1786 size_t fileWrite(File& f, const uint8_t* buf, size_t len) {
1787 Inode& in = f.inode;
1788 if (!(f.flags & OpenFlags::Write)) {
1789 fail(Error::ReadOnly);
1790 return 0;
1791 }
1792 if (!in.isReg()) {
1793 fail(Error::IsDir);
1794 return 0;

1798 return 0;

1802 size_t done = 0;

1805 uint32_t off = (uint32_t)(f.pos & (block_size - 1));

1816 break;

1819 ? f.goalPblk + 1

1829 f.goalPblk = pblk;

1833 f.map.pblk = 0;

1836 (off + n < block_size && !cache.fill(base + off + n, 0, block_size - off - n))) {
1837 fail(Error::IO);
1838 break;
1839 }
1840 }
1841 if (!cache.writeBulk(blockOff(pblk) + off, buf + done, n)) {
1842 fail(Error::IO);
1843 break;
1844 }
1845 done += n;
1846 f.pos += n;
1847 if (f.pos > in.size) {
1848 in.size = f.pos;
1849 f.dirty = true;
1850 }
1851 }
1852 if (done > 0) {
1853 in.mtime = in.ctime = now();
1854 f.dirty = true;
1855 }
1856 return done;
1857 }

1861 f.dirty = false;
1862 return writeInode(f.inode) && sync();
1863 }
