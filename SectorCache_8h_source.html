108 bool readBulk(uint64_t off, void* dst, size_t len) {
109 uint8_t* d = (uint8_t*)dst;
110 while (len > 0) {
111 if (off % SECTOR == 0 && len >= SECTOR) {
112 uint32_t lba = (uint32_t)(off / SECTOR);
113 uint32_t n = (uint32_t)(len / SECTOR);
114 if (n > TINYEXT4_MAX_TRANSFER_SECTORS) n = TINYEXT4_MAX_TRANSFER_SECTORS;
115 if (p_io->disk_read(0, d, base + lba, n) != RES_OK) return false;
116 // cached sectors may be newer than the device
117 for (auto& e : *this) {
118 if (e.valid && e.lba >= lba && e.lba < lba + n)
119 memcpy(d + (e.lba - lba) * SECTOR, e.data, SECTOR);
120 }
121 d += n * SECTOR;
122 off += (uint64_t)n * SECTOR;
123 len -= n * SECTOR;
124 } else {
125 size_t n = SECTOR - (size_t)(off % SECTOR);
126 if (n > len) n = len;
127 if (!read(off, d, n)) return false;
128 d += n;
129 off += n;
130 len -= n;
131 }
132 }
133 return true;
134 }

190 // zero initialized by the allocator
191 struct Entry {
192 uint32_t lba;
193 uint32_t stamp;
194 bool valid;
195 bool dirty;
196 uint8_t data[SECTOR];
197 };
198 Entry* entries = nullptr;
199 uint16_t count = TINYEXT4_CACHE_SECTORS;
200 Allocator* p_alloc = &defaultAllocator();
201 IO* p_io = nullptr;
202
203 // range based for over the entries
204 Entry* begin() { return entries; }
205 Entry* end() { return entries ? entries + count : nullptr; }
206 uint32_t base = 0;
207 uint32_t clock = 0;
208
209 bool writeEntry(Entry& e) {
210 if (p_io->disk_write(0, e.data, base + e.lba, 1) != RES_OK) return false;
211 e.dirty = false;
212 return true;
213 }
214
216 Entry* get(uint32_t lba, bool load) {
217 if (entries == nullptr) return nullptr;
218 Entry* victim = &entries[0];
219 for (auto& e : *this) {
220 if (e.valid && e.lba == lba) {
221 e.stamp = ++clock;
222 return &e;
223 }
224 if (!e.valid) {
225 if (victim->valid) victim = &e;
226 } else if (victim->valid && e.stamp < victim->stamp) {
227 victim = &e;
228 }
229 }
230 if (victim->valid && victim->dirty && !writeEntry(*victim)) return nullptr;
231 victim->valid = false;
232 victim->dirty = false;
233 if (load && p_io->disk_read(0, victim->data, base + lba, 1) != RES_OK)
234 return nullptr;
235 victim->lba = lba;
236 victim->valid = true;
237 victim->stamp = ++clock;
238 return victim;
239 }
240
241 bool modify(uint64_t off, size_t len, const uint8_t* src, uint8_t value) {
242 while (len > 0) {
243 uint32_t o = (uint32_t)(off % SECTOR);
244 size_t n = SECTOR - o < len ? SECTOR - o : len;
245 // a fully overwritten sector does not need to be read first
246 Entry* e = get((uint32_t)(off / SECTOR), n != SECTOR);
247 if (e == nullptr) return false;
248 if (src) {
249 memcpy(e->data + o, src, n);
250 src += n;
251 } else {
252 memset(e->data + o, value, n);
253 }
254 e->dirty = true;
255 off += n;
256 len -= n;
257 }
258 return true;
259 }
