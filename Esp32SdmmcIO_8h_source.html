133 bool begin(bool mode1bit, int max_freq_khz, sdmmc_host_t host_cfg,
134 sdmmc_slot_config_t slot_cfg) {
135 // Store configurations
136 host_config = host_cfg;
137 slot_config = slot_cfg;
138
139 // Set bus width
140 if (mode1bit) {
141 slot_config.width = 1;
142 } else {
143 slot_config.width = 4;
144 }
145
146 // Set maximum frequency
147 host_config.max_freq_khz = max_freq_khz;
148
149 // host_config.flags &= ~SDMMC_HOST_FLAG_DDR;
150
151 // Clean up any previous initialization state
152 // (in case SDMMC was used before and not properly cleaned up)
153 sdmmc_host_deinit();
154
155 // Initialize host peripheral
156 esp_err_t err = sdmmc_host_init();
157 if (err != ESP_OK) {
158 ESP_LOGE("Esp32SdmmcIO", "sdmmc_host_init failed: 0x%x", err);
159 stat = STA_NOINIT;
160 return false;
161 }
162
163 // Initialize slot (ESP32 only has SDMMC_HOST_SLOT_1)
164 err = sdmmc_host_init_slot(SDMMC_HOST_SLOT_1, &slot_config);
165 if (err != ESP_OK) {
166 ESP_LOGE("Esp32SdmmcIO", "sdmmc_host_init_slot failed: 0x%x", err);
167 sdmmc_host_deinit();
168 stat = STA_NOINIT;
169 return false;
170 }
171
172 // Allocate card structure
173 if (card == nullptr) {
174 card = (sdmmc_card_t*)malloc(sizeof(sdmmc_card_t));
175 if (card == nullptr) {
176 sdmmc_host_deinit();
177 stat = STA_NOINIT;
178 return false;
179 }
180 }
181 memset(card, 0, sizeof(sdmmc_card_t));
182
183 // Probe and initialize card
184 card->host = host_config;
185 err = sdmmc_card_init(&host_config, card);
186 if (err != ESP_OK) {
187 free(card);
188 card = nullptr;
189 sdmmc_host_deinit();
190 stat = STA_NODISK;
191 return false;
192 }
193
194 // Determine card type flags
195 CardType = 0;
196 if (card->is_mmc) {
197 CardType = CT_MMC;
198 } else {
199 // SD card
200 if (card->ocr & SD_OCR_SDHC_CAP) {
201 CardType = CT_SD2 | CT_BLOCK; // SDHC/SDXC
202 } else {
203 CardType = CT_SD2; // SDSC
204 }
205 }
206
207 stat = STA_CLEAR;
208 return true;
209 }

233 DSTATUS disk_initialize(BYTE drv) override {
234 if (drv != 0) return STA_NOINIT;
235
236 // If already initialized, return current status
237 if (stat == STA_CLEAR) return stat;
238
239 // Try to initialize with stored settings or defaults
240 if (auto_init && host_config.flags == 0) {
241 // Constructor with mode1bit and freq was used
242 if (begin(init_mode1bit, init_max_freq_khz)) {
243 return STA_CLEAR;
244 }
245 } else if (auto_init && host_config.flags != 0) {
246 // Constructor with full config was used
247 if (begin(init_mode1bit, init_max_freq_khz, host_config, slot_config)) {
248 return STA_CLEAR;
249 }
250 } else {
251 // Default constructor was used, try with defaults
252 if (begin()) {
253 return STA_CLEAR;
254 }
255 }

319#if FF_IO_USE_IOCTL
320 DRESULT disk_ioctl(BYTE drv, ioctl_cmd_t cmd, void* buff) override {
321 if (drv != 0) return RES_PARERR;
322 if (stat & STA_NOINIT) return RES_NOTRDY;
323
324 DRESULT res = RES_ERROR;
325
326 switch (cmd) {
327 case CTRL_SYNC:
328 // SD_MMC handles synchronization internally
329 res = RES_OK;
330 break;
331
332 case GET_SECTOR_COUNT:
333 if (buff && card) {
334 // Get total sectors from card capacity
335 uint64_t capacity =
336 ((uint64_t)card->csd.capacity) * card->csd.sector_size;
337 *(DWORD*)buff = (DWORD)(capacity / 512);
338 res = RES_OK;
339 }
340 break;
341
342 case GET_SECTOR_SIZE:
343 if (buff) {
344 // SD cards use 512-byte sectors for compatibility
345 *(WORD*)buff = 512;
346 res = RES_OK;
347 }
348 break;
349
350 case GET_BLOCK_SIZE:
351 if (buff && card) {
352 // Return erase block size in sectors
353 // Use the card's erase sector size if available
354 uint32_t erase_size = card->csd.sector_size;
355 if (erase_size < 512) erase_size = 512;
356 *(DWORD*)buff = erase_size / 512;
357 if (*(DWORD*)buff == 0) *(DWORD*)buff = 1;
358 res = RES_OK;
359 }
360 break;
361
362 case CTRL_TRIM:
363 // TRIM/ERASE not implemented via SD_MMC library
364 res = RES_OK; // Return OK but don't actually trim
365 break;
366
367 case MMC_GET_TYPE:
368 if (buff) {
369 *(BYTE*)buff = CardType;
370 res = RES_OK;
371 }
372 break;
373
374 case MMC_GET_CSD:
375 if (buff && card) {
376 // Copy CSD register data
377 memcpy(buff, &card->csd, sizeof(card->csd));
378 res = RES_OK;
379 }
380 break;
381
382 case MMC_GET_CID:
383 if (buff && card) {
384 // Copy CID register data
385 memcpy(buff, &card->cid, sizeof(card->cid));
386 res = RES_OK;
387 }
388 break;
389
390 case MMC_GET_OCR:
391 if (buff && card) {
392 // Copy OCR register
393 *(DWORD*)buff = card->ocr;
394 res = RES_OK;
395 }
396 break;
397
398 case MMC_GET_SDSTAT:
399 if (buff && card) {
400 // SD Status is only defined for SD cards (ACMD13)
401 if (card->is_mmc) {
402 res = RES_PARERR;
403 } else if (read_sd_status(buff) == ESP_OK) {
404 res = RES_OK;
405 } else {
406 res = RES_ERROR;
407 }
408 }
409 break;
410
411 default:
412 res = RES_PARERR;
413 break;
414 }
415
416 return res;
417 }

464 esp_err_t read_sd_status(void* buff) {
465 sdmmc_command_t cmd;
466
467 // CMD55: next command is an application specific command
468 memset(&cmd, 0, sizeof(cmd));
469 cmd.opcode = MMC_APP_CMD;
470 cmd.arg = (uint32_t)card->rca << 16;
471 cmd.flags = SCF_CMD_AC | SCF_RSP_R1;
472 esp_err_t err = card->host.do_transaction(card->host.slot, &cmd);
473 if (err != ESP_OK) return err;
474 if (cmd.error != ESP_OK) return cmd.error;
475
476 // ACMD13: read 64 bytes of SD status as a single data block
477 memset(&cmd, 0, sizeof(cmd));
478 cmd.opcode = SD_APP_SD_STATUS;
479 cmd.arg = 0;
480 cmd.flags = SCF_CMD_ADTC | SCF_RSP_R1;
481 cmd.data = buff;
482 cmd.datalen = 64;
483 cmd.blklen = 64;
484 err = card->host.do_transaction(card->host.slot, &cmd);
485 if (err != ESP_OK) return err;
486 return cmd.error;
487 }
