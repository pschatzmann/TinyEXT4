66 UINT count) override {
67 if (pdrv != 0) return RES_NOTRDY;
68 if (status == STA_NOINIT) return RES_NOTRDY;
69 if (read_only) return RES_WRPRT;
70 if (seek_to(file, (uint64_t)sector * sector_size) != 0) return RES_ERROR;

75 DRESULT disk_ioctl(BYTE pdrv, ioctl_cmd_t cmd, void* buff) override {
76 if (pdrv != 0) return RES_PARERR;
77 switch (cmd) {
78 case CTRL_SYNC:
79 return fflush(file) == 0 ? RES_OK : RES_ERROR;
80
81 case GET_SECTOR_COUNT: {
82 if (buff == nullptr) return RES_PARERR;
83 DWORD result = (DWORD)sector_count;
84 memcpy(buff, &result, sizeof(result));
85 return RES_OK;
86 }
87
88 case GET_SECTOR_SIZE: {
89 if (buff == nullptr) return RES_PARERR;
90 WORD result = (WORD)sector_size;
91 memcpy(buff, &result, sizeof(result));
92 return RES_OK;
93 }
94
95 case GET_BLOCK_SIZE: {
96 if (buff == nullptr) return RES_PARERR;
97 DWORD result = 1;
98 memcpy(buff, &result, sizeof(result));
99 return RES_OK;
100 }
101
102 default:

104 }

124 bool open_or_create() {
125 read_only = false;
126 file = fopen(path, "r+b");
127 if (file == nullptr) {
128 file = fopen(path, "rb");
129 read_only = (file != nullptr);
130 }
131 if (file == nullptr) {
132 if (sector_count == 0) return false;
133 file = fopen(path, "w+b");
134 if (file == nullptr) return false;
135 }
136 // grow (sparsely) to the requested size if the file is smaller, whether
137 // freshly created or a pre-existing image that's too small
138 uint64_t wanted_size = (uint64_t)sector_count * sector_size;
139 if (seek_to(file, 0) != 0) return false;
140 if (fseek(file, 0, SEEK_END) != 0) return false;
141#ifdef _WIN32
142 __int64 current_size = _ftelli64(file);
143#else
144 off_t current_size = ftello(file);
145#endif
146 if (current_size < 0) return false;
147 if (sector_count == 0) {
148 sector_count = (size_t)((uint64_t)current_size / sector_size);
149 return sector_count > 0;
150 }
151 if (!read_only && (uint64_t)current_size < wanted_size) {
152 if (seek_to(file, wanted_size - 1) != 0) return false;
153 uint8_t zero = 0;
154 if (fwrite(&zero, 1, 1, file) != 1) return false;
155 fflush(file);
156 }
157 return true;
158 }
