65 DRESULT disk_ioctl(BYTE pdrv, ioctl_cmd_t cmd, void* buff) override {
66 DRESULT res;
67 if (pdrv) return RES_PARERR; /* Check parameter */
68 if (status == STA_NOINIT) return RES_NOTRDY;
69 if (buff == nullptr && cmd != CTRL_SYNC) return RES_PARERR;
70 switch (cmd) {
71 case CTRL_SYNC: // Wait for end of internal write process of the drive
72 res = RES_OK;
73 p_stream->flush();
74 break;
75
76 case GET_SECTOR_COUNT: { // Get drive capacity in unit of sector (DWORD)
77 DWORD result = p_stream->sectorCount();
78 memcpy(buff, &result, (sizeof(result)));
79 res = RES_OK;
80 } break;
81
82 case GET_BLOCK_SIZE: { // Get erase block size in unit of sector (DWORD)
83 DWORD result = 1;
84 memcpy(buff, &result, (sizeof(result)));
85 res = RES_OK;
86 } break;
87
88 case CTRL_TRIM: { // Erase a block of sectors (used when _USE_ERASE == 1)
89 DWORD range[2];
90 // determine range
91 memcpy(range, buff, sizeof(range));
92 if (range[0] > range[1]) return RES_PARERR;
93 // clear memory
94 p_stream->eraseSector(range[0], range[1]);
95 res = RES_OK;
96 } break;
97
98 default:
99 res = RES_PARERR;
100 break;
101 }
102 return res;
103 }
