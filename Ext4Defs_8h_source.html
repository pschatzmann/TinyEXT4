28inline const char* errorText(Error e) {
29 switch (e) {
30 case Error::OK: return "OK";
31 case Error::IO: return "IO error";
32 case Error::NotMounted: return "not mounted";
33 case Error::NotExt4: return "not an ext4 file system";
34 case Error::Unsupported: return "unsupported";
35 case Error::NotFound: return "not found";
36 case Error::Exists: return "exists";
37 case Error::NotDir: return "not a directory";
38 case Error::IsDir: return "is a directory";
39 case Error::NotEmpty: return "directory not empty";
40 case Error::NoSpace: return "no space";
41 case Error::ReadOnly: return "read only";
42 case Error::Invalid: return "invalid argument";
43 case Error::Corrupt: return "corrupt file system";
44 case Error::NoMemory: return "out of memory";
45 }
46 return "?";
47}
