76 DSTATUS disk_initialize(BYTE drv /* Physical drive number (0) */
77 ) override {
78 BYTE n, cmd, ty, ocr[4];
79
80 if (drv != 0) return STA_NOINIT; /* Supports only drive 0 */
81 // assume SPI already init init_spi(); /* Initialize SPI */
82
83 if (stat & STA_NODISK) return stat; /* Is card existing in the soket? */
84
85 set_spi_fast(false);
86 for (n = 10; n; n--) xchg_spi(0xFF); /* Send 80 dummy clocks */
87
88 ty = 0;
89 if (send_cmd(CMD0, 0) == 1) { /* Put the card SPI/Idle state */
90 spi_timer_on(1000); /* Initialization timeout = 1 sec */
91 if (send_cmd(CMD8, 0x1AA) == 1) { /* SDv2? */
92 for (n = 0; n < 4; n++)
93 ocr[n] = xchg_spi(0xFF); /* Get 32 bit return value of R7 resp */
94 if (ocr[2] == 0x01 &&
95 ocr[3] == 0xAA) { /* Is the card supports vcc of 2.7-3.6V? */
96 while (spi_timer_status() &&
97 send_cmd(ACMD41, 1UL << 30)); /* Wait for end of initialization
98 with ACMD41(HCS) */
99 if (spi_timer_status() &&
100 send_cmd(CMD58, 0) == 0) { /* Check CCS bit in the OCR */
101 for (n = 0; n < 4; n++) ocr[n] = xchg_spi(0xFF);
102 ty =
103 (ocr[0] & 0x40) ? CT_SD2 | CT_BLOCK : CT_SD2; /* Card id SDv2 */
104 }
105 }
106 } else { /* Not SDv2 card */
107 if (send_cmd(ACMD41, 0) <= 1) { /* SDv1 or MMC? */
108 ty = CT_SD1;
109 cmd = ACMD41; /* SDv1 (ACMD41(0)) */
110 } else {
111 ty = CT_MMC;
112 cmd = CMD1; /* MMCv3 (CMD1(0)) */
113 }
114 while (spi_timer_status() &&
115 send_cmd(cmd, 0)); /* Wait for end of initialization */
116 if (!spi_timer_status() ||
117 send_cmd(CMD16, 512) != 0) /* Set block length: 512 */
118 ty = 0;
119 }
120 }
121 CardType = ty; /* Card type */
122 despiselect();
123
124 if (ty) { /* OK */
125 set_spi_fast(true); /* Set fast clock */
126 stat = STA_CLEAR; /* Clear STA_NOINIT flag */
127 } else { /* Failed */
128 stat = STA_NOINIT;
129 }
130
131 return stat;
132 }

148 DRESULT disk_read(
149 BYTE drv, /* Physical drive number (0) */
150 BYTE *buff, /* Pointer to the data buffer to store read data */
151 DWORD sector, /* Start sector number (LBA) */
152 UINT count /* Number of sectors to read (1..128) */
153 ) override {
154 if (drv || !count) return RES_PARERR; /* Check parameter */
155 if (stat & STA_NOINIT) return RES_NOTRDY; /* Check if drive is ready */
156
157 if (!(CardType & CT_BLOCK))
158 sector *= 512; /* LBA ot BA conversion (byte addressing cards) */
159
160 if (count == 1) { /* Single sector read */
161 if ((send_cmd(CMD17, sector) == 0) /* READ_SINGLE_BLOCK */
162 && rcvr_datablock(buff, 512)) {
163 count = 0;
164 }
165 } else { /* Multiple sector read */
166 if (send_cmd(CMD18, sector) == 0) { /* READ_MULTIPLE_BLOCK */
167 do {
168 if (!rcvr_datablock(buff, 512)) break;
169 buff += 512;
170 } while (--count);
171 send_cmd(CMD12, 0); /* STOP_TRANSMISSION */
172 wait_ready(500); /* Wait for card to be ready after stop transmission */
173 }
174 }
175 despiselect();
176
177 return count ? RES_ERROR : RES_OK; /* Return result */
178 }

224#if FF_IO_USE_IOCTL
225 DRESULT disk_ioctl(BYTE drv, /* Physical drive number (0) */
226 ioctl_cmd_t cmd, /* Control command code */
227 void *buff /* Pointer to the conrtol data */
228 ) override {
229 DRESULT res;
230 BYTE n, csd[16];
231 DWORD *dp, st, ed, csize;
232
233 if (drv) return RES_PARERR; /* Check parameter */
234 if (stat & STA_NOINIT) return RES_NOTRDY; /* Check if drive is ready */
235 if (buff == nullptr && cmd != CTRL_SYNC) return RES_PARERR; /* No buffer */
236
237 res = RES_ERROR;
238
239 switch (cmd) {
240 case CTRL_SYNC: /* Wait for end of internal write process of the drive */
241 if (spiselect()) res = RES_OK;
242 break;
243
244 case GET_SECTOR_COUNT: /* Get drive capacity in unit of sector (DWORD) */
245 if ((send_cmd(CMD9, 0) == 0) && rcvr_datablock(csd, 16)) {
246 if ((csd[0] >> 6) == 1) { /* SDC ver 2.00 */
247 csize =
248 csd[9] + ((WORD)csd[8] << 8) + ((DWORD)(csd[7] & 63) << 16) + 1;
249 *(DWORD *)buff = csize << 10;
250 } else { /* SDC ver 1.XX or MMC ver 3 */
251 n = (csd[5] & 15) + ((csd[10] & 128) >> 7) + ((csd[9] & 3) << 1) +
252 2;
253 csize = (csd[8] >> 6) + ((WORD)csd[7] << 2) +
254 ((WORD)(csd[6] & 3) << 10) + 1;
255 *(DWORD *)buff = csize << (n - 9);
256 }
257 res = RES_OK;
258 }
259 break;
260
261 case GET_BLOCK_SIZE: /* Get erase block size in unit of sector (DWORD) */
262 if (CardType & CT_SD2) { /* SDC ver 2.00 */
263 if (send_cmd(ACMD13, 0) == 0) { /* Read SD status */
264 xchg_spi(0xFF);
265 if (rcvr_datablock(csd, 16)) { /* Read partial block */
266 for (n = 64 - 16; n; n--)
267 xchg_spi(0xFF); /* Purge trailing data */
268 *(DWORD *)buff = 16UL << (csd[10] >> 4);
269 res = RES_OK;
270 }
271 }
272 } else { /* SDC ver 1.XX or MMC */
273 if ((send_cmd(CMD9, 0) == 0) &&
274 rcvr_datablock(csd, 16)) { /* Read CSD */
275 if (CardType & CT_SD1) { /* SDC ver 1.XX */
276 *(DWORD *)buff =
277 (((csd[10] & 63) << 1) + ((WORD)(csd[11] & 128) >> 7) + 1)
278 << ((csd[13] >> 6) - 1);
279 } else { /* MMC */
280 *(DWORD *)buff =
281 ((WORD)((csd[10] & 124) >> 2) + 1) *
282 (((csd[11] & 3) << 3) + ((csd[11] & 224) >> 5) + 1);
283 }
284 res = RES_OK;
285 }
286 }
287 break;
288
289 case CTRL_TRIM: /* Erase a block of sectors (used when _USE_ERASE ==
290 1) */
291 if (!(CardType & CT_SDC)) break; /* Check if the card is SDC */
292 if (disk_ioctl(drv, MMC_GET_CSD, csd)) break; /* Get CSD */
293 if (!(csd[0] >> 6) && !(csd[10] & 0x40))
294 break; /* Check if sector erase can be applied to the card */
295 dp = (DWORD *)buff;
296 st = dp[0];
297 ed = dp[1]; /* Load sector block */
298 if (!(CardType & CT_BLOCK)) {
299 st *= 512;
300 ed *= 512;
301 }
302 if (send_cmd(CMD32, st) == 0 && send_cmd(CMD33, ed) == 0 &&
303 send_cmd(CMD38, 0) == 0 &&
304 wait_ready(30000)) { /* Erase sector block */
305 res = RES_OK;
306 }
307 break;
308
309 default:
310 res = RES_PARERR;
311 }
312
313 despiselect();
314
315 return res;
316 }

345 /* Receive multiple byte */
346 void rcvr_spi_multi(BYTE *buff, UINT btr) {
347 // SPIClass::transfer(buf, len) transmits buf's own contents while
348 // overwriting it with the received bytes, so pre-fill with the dummy
349 // 0xFF clock-out byte, then do the whole block in one call instead of
350 // looping single-byte transfers - lets the SPI HAL/DMA burst the
351 // transfer instead of paying per-byte call overhead each time.
352 memset(buff, 0xFF, btr);
353 p_spi->transfer(buff, btr);

400 /* Receive a data packet from the MMC */
401
402 int rcvr_datablock( /* 1:OK, 0:Error */
403 BYTE *buff, /* Data buffer */
404 UINT btr /* Data block length (byte) */
405 ) {
406 BYTE token;
407
408 uint64_t end = millis() + 200;
409 do { /* Wait for DataStart token in timeout of 200ms */
410 token = xchg_spi(0xFF);
411 /* This loop will take a time. Insert rot_rdq() here for multitask
412 * envilonment. */
413 } while ((token == 0xFF) && millis() < end);
414 if (token != 0xFE)
415 return 0; /* Function fails if invalid DataStart token or timeout */
416
417 rcvr_spi_multi(buff, btr); /* Receive data from card */
418 xchg_spi(0xFF);
419 xchg_spi(0xFF); /* Discard CRC */
420
421 return 1; /* Function succeeded */
422 }

449 /* Send a command packet to the MMC */
450 BYTE send_cmd( /* Return value: R1 resp (bit7==1:Failed to send) */
451 BYTE cmd, /* Command index */
452 DWORD arg /* Argument */
453 ) {
454 BYTE n, res;
455
456 if (cmd & 0x80) { /* Send a CMD55 prior to ACMD<n> */
457 cmd &= 0x7F;
458 res = send_cmd(CMD55, 0);
459 if (res > 1) return res;
460 }
461
462 /* Select the card and wait for ready except to stop multiple block read */
463 if (cmd != CMD12) {
464 despiselect();
465 if (!spiselect()) return 0xFF;
466 }
467
468 /* Send command packet */
469 xchg_spi(0x40 | cmd); /* Start + command index */
470 xchg_spi((BYTE)(arg >> 24)); /* Argument[31..24] */
471 xchg_spi((BYTE)(arg >> 16)); /* Argument[23..16] */
472 xchg_spi((BYTE)(arg >> 8)); /* Argument[15..8] */
473 xchg_spi((BYTE)arg); /* Argument[7..0] */
474 n = 0x01; /* Dummy CRC + Stop */
475 if (cmd == CMD0) n = 0x95; /* Valid CRC for CMD0(0) */
476 if (cmd == CMD8) n = 0x87; /* Valid CRC for CMD8(0x1AA) */
477 xchg_spi(n);
478
479 /* Receive command resp */
480 if (cmd == CMD12)
481 xchg_spi(0xFF); /* Diacard following one byte when CMD12 */
482 n = 10; /* Wait for response (10 bytes max) */
483 do {
484 res = xchg_spi(0xFF);
485 } while ((res & 0x80) && --n);
486
487 return res; /* Return received response */
488 }
