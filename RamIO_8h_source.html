35 bool load(const uint8_t* image, size_t len) {
36 if (disk_initialize(0) != STA_CLEAR) return false;
37 if (len > (size_t)sector_count * sector_size) return false;
38 for (size_t pos = 0; pos < len; pos += sector_size) {
39 size_t n = len - pos < (size_t)sector_size ? len - pos : sector_size;
40 memcpy(sectors[pos / sector_size], image + pos, n);
41 }
42 return true;
43 }
44
45 DSTATUS disk_initialize(BYTE pdrv) override {
46 if (pdrv != 0) return STA_NODISK;
47 // allocate sectors
48 if (sectors.size() == 0) {
49 for (size_t j = 0; j < sector_count; j++) {
50 uint8_t* ptr = (uint8_t*)p_alloc->allocate(sector_size);
51 if (ptr == nullptr) {
52 // out of memory: release what we got and report the failure
53 for (auto* p : sectors) p_alloc->release(p);
54 sectors.clear();
55 status = STA_NOINIT;
56 return status;
57 }
58 sectors.push_back(ptr);
59 }
60 }
61 status = STA_CLEAR;
62 return status;
63 }
64
65 DSTATUS disk_status(BYTE pdrv) override {
66 if (pdrv != 0) return STA_NODISK;
67 return status;
68 }
69
70 DRESULT disk_read(BYTE pdrv, BYTE* buffer, LBA_t sectorNo, UINT sectorCount) override {
71 if (pdrv != 0) return RES_NOTRDY;
72 if (status == STA_NOINIT) return RES_NOTRDY;
73 if (sectors.size() == 0 || sectorNo + sectorCount > sector_count) return RES_ERROR;
74 for (LBA_t j = sectorNo; j < sectorNo + sectorCount; j++) {
75 int idx = j - sectorNo;
76 uint8_t* source = sectors[j];
77 memcpy(buffer + (idx * sector_size), source , sector_size);
78 }
79 return RES_OK;
80 }
81
82 DRESULT disk_write(BYTE pdrv, const BYTE* buffer, LBA_t sectorNo,
83 UINT sectorCount) override {
84 if (pdrv != 0) return RES_NOTRDY;
85 if (status == STA_NOINIT) return RES_NOTRDY;
86 if (sectors.size() == 0 || sectorNo + sectorCount > sector_count) return RES_ERROR;
87 for (LBA_t j = sectorNo; j < sectorNo + sectorCount; j++) {
88 int idx = j - sectorNo;
89 uint8_t* target = sectors[j];
90 memcpy(target, buffer + (idx * sector_size), sector_size);
91 }
92 return RES_OK;
93 }
94
95 DRESULT disk_ioctl(BYTE pdrv, ioctl_cmd_t cmd, void* buffer) override {
96 DRESULT res;
97 if (pdrv) return RES_PARERR; /* Check parameter */
98 switch (cmd) {
99 case CTRL_SYNC: /* Wait for end of internal write process of the drive */
100 res = RES_OK;
101 break;
102
103 case GET_SECTOR_COUNT: { /* Get drive capacity in unit of sector (DWORD)
104 */
105 DWORD result = sector_count;
106 memcpy(buffer, &result, (sizeof(result)));
107 res = RES_OK;
108 } break;
109
110 case GET_BLOCK_SIZE: { /* Get erase block size in unit of sector (DWORD)
111 */
112 DWORD result = 1;
113 memcpy(buffer, &result, (sizeof(result)));
114 res = RES_OK;
115 } break;
116
117 case CTRL_TRIM: { /* Erase a block of sectors (used when _USE_ERASE == 1)
118 */
119 if (buffer == nullptr || status == STA_NOINIT) {
120 res = RES_PARERR;
121 break;
122 }
123 DWORD range[2];
124 // determine range
125 memcpy(&range, buffer, sizeof(range));
126 // validate range: must be inside the drive and allocated
127 if (range[0] > range[1] || range[1] >= sector_count ||
128 sectors.size() != sector_count) {
129 res = RES_PARERR;
130 break;
131 }
132 // clear memory
133 for (DWORD j = range[0]; j <= range[1]; j++) {
134 memset(sectors[j], 0, sector_size);
135 }
136 res = RES_OK;
137 } break;
138
139 default:
140 res = RES_PARERR;
141 break;
142 }
143 return res;
144 }
145
146 protected:
147 std::vector<uint8_t*, AllocatorExt<uint8_t*>> sectors;
148 Allocator* p_alloc = nullptr;
149 DSTATUS status = STA_NOINIT;
150 int sector_size = 512;
151 size_t sector_count = 0;
152};
